-- Create Database
CREATE DATABASE clarity_mental_health;
USE clarity_mental_health;

-- 1. USERS TABLE (Strong Entity)
CREATE TABLE Users (
    user_id INT PRIMARY KEY AUTO_INCREMENT,           -- ◉ PK
    email VARCHAR(255) UNIQUE NOT NULL,               -- ○ Unique
    password_hash VARCHAR(255) NOT NULL,              -- ○
    first_name VARCHAR(100) NOT NULL,                 -- ○
    last_name VARCHAR(100),                           -- ○
    date_of_birth DATE,                               -- ○
    gender ENUM('male', 'female', 'other', 'prefer_not_to_say'),
    phone VARCHAR(20),                                -- ○
    profile_picture VARCHAR(500),                     -- ○
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,   -- ○
    last_login TIMESTAMP                              -- ○
);

-- 2. ASSESSMENTS TABLE (Strong Entity)
CREATE TABLE Assessments (
    assessment_id INT PRIMARY KEY AUTO_INCREMENT,     -- ◉ PK
    name VARCHAR(100) NOT NULL,                       -- ○
    full_name VARCHAR(255),                           -- ○
    description TEXT,                                 -- ○
    category ENUM('anxiety', 'depression', 'stress', 'ptsd', 'ocd', 
                  'bipolar', 'substance', 'sleep'),   -- ○
    total_questions INT,                              -- ○
    max_score INT,                                    -- ○
    time_limit_minutes INT                            -- ○
);

-- 3. ASSESSMENT_QUESTIONS TABLE (Weak Entity - Double Rectangle)
CREATE TABLE AssessmentQuestions (
    question_id INT PRIMARY KEY AUTO_INCREMENT,       -- ◉ PK
    assessment_id INT NOT NULL,                       -- ◉ FK (Identifying)
    question_number INT NOT NULL,                     -- ○ Partial Key
    question_text TEXT NOT NULL,                      -- ○
    max_points INT DEFAULT 3,                         -- ○
    FOREIGN KEY (assessment_id) REFERENCES Assessments(assessment_id)
        ON DELETE CASCADE                             -- Identifying Relationship
);

-- 4. SEVERITY_LEVELS TABLE (Strong Entity)
CREATE TABLE SeverityLevels (
    severity_id INT PRIMARY KEY AUTO_INCREMENT,       -- ◉ PK
    assessment_id INT NOT NULL,                       -- ◉ FK
    min_score INT NOT NULL,                           -- ○
    max_score INT NOT NULL,                           -- ○
    severity_label VARCHAR(50),                       -- ○
    description TEXT,                                 -- ○
    recommendation TEXT,                              -- ○
    requires_professional BOOLEAN DEFAULT FALSE,      -- ○
    FOREIGN KEY (assessment_id) REFERENCES Assessments(assessment_id)
        ON DELETE CASCADE
);

-- 5. USER_ASSESSMENTS TABLE (Associative Entity)
CREATE TABLE UserAssessments (
    user_assessment_id INT PRIMARY KEY AUTO_INCREMENT, -- ◉ PK
    user_id INT NOT NULL,                              -- ◉ FK
    assessment_id INT NOT NULL,                        -- ◉ FK
    total_score INT,                                   -- ○
    severity_id INT,                                   -- ◉ FK
    taken_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,      -- ○
    completion_time_seconds INT,                       -- ○
    is_completed BOOLEAN DEFAULT FALSE,                -- ○
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (assessment_id) REFERENCES Assessments(assessment_id),
    FOREIGN KEY (severity_id) REFERENCES SeverityLevels(severity_id)
);

-- 6. USER_ASSESSMENT_RESPONSES TABLE (Weak Entity - Double Rectangle)
CREATE TABLE UserAssessmentResponses (
    response_id INT PRIMARY KEY AUTO_INCREMENT,        -- ◉ PK
    user_assessment_id INT NOT NULL,                   -- ◉ FK (Identifying)
    question_id INT NOT NULL,                          -- ◉ FK
    response_value INT CHECK (response_value >= 0 AND response_value <= 3),
    response_text VARCHAR(100),                        -- ○
    FOREIGN KEY (user_assessment_id) REFERENCES UserAssessments(user_assessment_id)
        ON DELETE CASCADE,                             -- Identifying Relationship
    FOREIGN KEY (question_id) REFERENCES AssessmentQuestions(question_id)
);

-- 7. DIAGNOSES TABLE (Strong Entity)
CREATE TABLE Diagnoses (
    diagnosis_id INT PRIMARY KEY AUTO_INCREMENT,       -- ◉ PK
    name VARCHAR(200) NOT NULL,                        -- ○
    code VARCHAR(20),                                  -- ○ (ICD-10)
    category VARCHAR(100),                             -- ○
    description TEXT,                                  -- ○
    symptoms TEXT,                                     -- ○
    treatment_options TEXT                             -- ○
);

-- 8. ASSESSMENT_DIAGNOSES TABLE (Junction/Associative Entity for M:N)
CREATE TABLE AssessmentDiagnoses (
    assessment_id INT NOT NULL,                        -- ◉ PFK
    diagnosis_id INT NOT NULL,                         -- ◉ PFK
    PRIMARY KEY (assessment_id, diagnosis_id),         -- Composite PK
    FOREIGN KEY (assessment_id) REFERENCES Assessments(assessment_id),
    FOREIGN KEY (diagnosis_id) REFERENCES Diagnoses(diagnosis_id)
);

-- 9. USER_DIAGNOSES TABLE (Associative Entity)
CREATE TABLE UserDiagnoses (
    user_diagnosis_id INT PRIMARY KEY AUTO_INCREMENT,  -- ◉ PK
    user_id INT NOT NULL,                              -- ◉ FK
    diagnosis_id INT NOT NULL,                         -- ◉ FK
    user_assessment_id INT,                            -- ◉ FK (optional)
    severity ENUM('mild', 'moderate', 'severe'),       -- ○
    diagnosed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,  -- ○
    is_active BOOLEAN DEFAULT TRUE,                    -- ○
    notes TEXT,                                        -- ○
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (diagnosis_id) REFERENCES Diagnoses(diagnosis_id),
    FOREIGN KEY (user_assessment_id) REFERENCES UserAssessments(user_assessment_id)
);

-- 10. THERAPISTS TABLE (Strong Entity)
CREATE TABLE Therapists (
    therapist_id INT PRIMARY KEY AUTO_INCREMENT,       -- ◉ PK
    user_id INT NOT NULL UNIQUE,                       -- ◉ FK (1:1 with Users)
    license_number VARCHAR(100) UNIQUE NOT NULL,       -- ○ Unique
    specialization VARCHAR(255),                       -- ○
    years_experience INT,                              -- ○
    hourly_rate DECIMAL(10, 2),                        -- ○
    bio TEXT,                                          -- ○
    is_verified BOOLEAN DEFAULT FALSE,                 -- ○
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE
);

-- 11. APPOINTMENTS TABLE (Associative Entity)
CREATE TABLE Appointments (
    appointment_id INT PRIMARY KEY AUTO_INCREMENT,     -- ◉ PK
    user_id INT NOT NULL,                              -- ◉ FK
    therapist_id INT NOT NULL,                         -- ◉ FK
    scheduled_at DATETIME NOT NULL,                    -- ○
    duration_minutes INT DEFAULT 60,                   -- ○
    status ENUM('scheduled', 'completed', 'cancelled', 'no_show') 
           DEFAULT 'scheduled',                        -- ○
    notes TEXT,                                        -- ○
    meeting_link VARCHAR(500),                         -- ○
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (therapist_id) REFERENCES Therapists(therapist_id)
);

-- 12. MOOD_ENTRIES TABLE (Strong Entity)
CREATE TABLE MoodEntries (
    entry_id INT PRIMARY KEY AUTO_INCREMENT,           -- ◉ PK
    user_id INT NOT NULL,                              -- ◉ FK
    mood_level INT CHECK (mood_level >= 1 AND mood_level <= 10),
    mood_type ENUM('happy', 'sad', 'anxious', 'calm', 'angry', 'neutral',
                   'excited', 'stressed', 'hopeful', 'tired'),
    notes TEXT,                                        -- ○
    entry_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,    -- ○
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE
);

-- 13. JOURNAL_ENTRIES TABLE (Strong Entity)
CREATE TABLE JournalEntries (
    journal_id INT PRIMARY KEY AUTO_INCREMENT,         -- ◉ PK
    user_id INT NOT NULL,                              -- ◉ FK
    title VARCHAR(255),                                -- ○
    content TEXT NOT NULL,                             -- ○
    is_private BOOLEAN DEFAULT TRUE,                   -- ○
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,    -- ○
    updated_at TIMESTAMP,                              -- ○
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE
);

-- 14. GOALS TABLE (Strong Entity)
CREATE TABLE Goals (
    goal_id INT PRIMARY KEY AUTO_INCREMENT,            -- ◉ PK
    user_id INT NOT NULL,                              -- ◉ FK
    title VARCHAR(255) NOT NULL,                       -- ○
    description TEXT,                                  -- ○
    target_date DATE,                                  -- ○
    status ENUM('active', 'completed', 'abandoned') DEFAULT 'active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,    -- ○
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE
);

-- 15. EMERGENCY_CONTACTS TABLE (Strong Entity)
CREATE TABLE EmergencyContacts (
    contact_id INT PRIMARY KEY AUTO_INCREMENT,         -- ◉ PK
    user_id INT NOT NULL,                              -- ◉ FK
    name VARCHAR(200) NOT NULL,                        -- ○
    relationship VARCHAR(100),                         -- ○
    phone VARCHAR(20) NOT NULL,                        -- ○
    is_primary BOOLEAN DEFAULT FALSE,                  -- ○
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE
);

-- 16. RESOURCES TABLE (Strong Entity)
CREATE TABLE Resources (
    resource_id INT PRIMARY KEY AUTO_INCREMENT,        -- ◉ PK
    title VARCHAR(255) NOT NULL,                       -- ○
    description TEXT,                                  -- ○
    resource_type ENUM('article', 'video', 'audio', 'exercise', 'meditation'),
    content_url VARCHAR(500),                          -- ○
    category VARCHAR(100),                             -- ○
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP     -- ○
);

-- 17. USER_RESOURCES TABLE (Junction/Associative Entity for M:N)
CREATE TABLE UserResources (
    user_id INT NOT NULL,                              -- ◉ PFK
    resource_id INT NOT NULL,                          -- ◉ PFK
    is_bookmarked BOOLEAN DEFAULT FALSE,               -- ○ (Relationship attribute)
    completed_at TIMESTAMP,                            -- ○ (Relationship attribute)
    PRIMARY KEY (user_id, resource_id),                -- Composite PK
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (resource_id) REFERENCES Resources(resource_id) ON DELETE CASCADE
);

-- Create Database
CREATE DATABASE clarity_mental_health;
USE clarity_mental_health;

-- 1. USERS TABLE
CREATE TABLE Users (
    user_id INT PRIMARY KEY AUTO_INCREMENT,
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100),
    date_of_birth DATE,
    gender ENUM('male', 'female', 'other', 'prefer_not_to_say'),
    phone VARCHAR(20),
    profile_picture VARCHAR(500),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    last_login TIMESTAMP
);

-- 2. ASSESSMENTS TABLE
CREATE TABLE Assessments (
    assessment_id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    full_name VARCHAR(255),
    description TEXT,
    category ENUM('anxiety', 'depression', 'stress', 'ptsd', 'ocd', 'bipolar', 
                  'substance', 'sleep'),
    total_questions INT,
    max_score INT,
    time_limit_minutes INT
);

-- 3. ASSESSMENT_QUESTIONS TABLE
CREATE TABLE AssessmentQuestions (
    question_id INT PRIMARY KEY AUTO_INCREMENT,
    assessment_id INT NOT NULL,
    question_number INT NOT NULL,
    question_text TEXT NOT NULL,
    max_points INT DEFAULT 3,
    FOREIGN KEY (assessment_id) REFERENCES Assessments(assessment_id)
        ON DELETE CASCADE
);

-- 4. SEVERITY_LEVELS TABLE
CREATE TABLE SeverityLevels (
    severity_id INT PRIMARY KEY AUTO_INCREMENT,
    assessment_id INT NOT NULL,
    min_score INT NOT NULL,
    max_score INT NOT NULL,
    severity_label VARCHAR(50),
    description TEXT,
    recommendation TEXT,
    requires_professional BOOLEAN DEFAULT FALSE,
    FOREIGN KEY (assessment_id) REFERENCES Assessments(assessment_id)
        ON DELETE CASCADE
);

-- 5. USER_ASSESSMENTS TABLE
CREATE TABLE UserAssessments (
    user_assessment_id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    assessment_id INT NOT NULL,
    total_score INT,
    severity_id INT,
    taken_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    completion_time_seconds INT,
    is_completed BOOLEAN DEFAULT FALSE,
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (assessment_id) REFERENCES Assessments(assessment_id),
    FOREIGN KEY (severity_id) REFERENCES SeverityLevels(severity_id)
);

-- 6. USER_ASSESSMENT_RESPONSES TABLE
CREATE TABLE UserAssessmentResponses (
    response_id INT PRIMARY KEY AUTO_INCREMENT,
    user_assessment_id INT NOT NULL,
    question_id INT NOT NULL,
    response_value INT CHECK (response_value >= 0 AND response_value <= 3),
    response_text VARCHAR(100),
    FOREIGN KEY (user_assessment_id) REFERENCES UserAssessments(user_assessment_id)
        ON DELETE CASCADE,
    FOREIGN KEY (question_id) REFERENCES AssessmentQuestions(question_id)
);

-- 7. DIAGNOSES TABLE
CREATE TABLE Diagnoses (
    diagnosis_id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(200) NOT NULL,
    code VARCHAR(20),
    category VARCHAR(100),
    description TEXT,
    symptoms TEXT,
    treatment_options TEXT
);

-- 8. ASSESSMENT_DIAGNOSES TABLE (Junction)
CREATE TABLE AssessmentDiagnoses (
    assessment_id INT NOT NULL,
    diagnosis_id INT NOT NULL,
    PRIMARY KEY (assessment_id, diagnosis_id),
    FOREIGN KEY (assessment_id) REFERENCES Assessments(assessment_id),
    FOREIGN KEY (diagnosis_id) REFERENCES Diagnoses(diagnosis_id)
);

-- 9. USER_DIAGNOSES TABLE
CREATE TABLE UserDiagnoses (
    user_diagnosis_id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    diagnosis_id INT NOT NULL,
    user_assessment_id INT,
    severity ENUM('mild', 'moderate', 'severe'),
    diagnosed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    is_active BOOLEAN DEFAULT TRUE,
    notes TEXT,
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (diagnosis_id) REFERENCES Diagnoses(diagnosis_id),
    FOREIGN KEY (user_assessment_id) REFERENCES UserAssessments(user_assessment_id)
);

-- 10. THERAPISTS TABLE
CREATE TABLE Therapists (
    therapist_id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL UNIQUE,
    license_number VARCHAR(100) UNIQUE NOT NULL,
    specialization VARCHAR(255),
    years_experience INT,
    hourly_rate DECIMAL(10, 2),
    bio TEXT,
    is_verified BOOLEAN DEFAULT FALSE,
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE
);

-- 11. APPOINTMENTS TABLE
CREATE TABLE Appointments (
    appointment_id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    therapist_id INT NOT NULL,
    scheduled_at DATETIME NOT NULL,
    duration_minutes INT DEFAULT 60,
    status ENUM('scheduled', 'completed', 'cancelled', 'no_show') DEFAULT 'scheduled',
    notes TEXT,
    meeting_link VARCHAR(500),
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (therapist_id) REFERENCES Therapists(therapist_id)
);

-- 12. MOOD_ENTRIES TABLE
CREATE TABLE MoodEntries (
    entry_id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    mood_level INT CHECK (mood_level >= 1 AND mood_level <= 10),
    mood_type ENUM('happy', 'sad', 'anxious', 'calm', 'angry', 'neutral',
                   'excited', 'stressed', 'hopeful', 'tired'),
    notes TEXT,
    entry_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE
);

-- 13. JOURNAL_ENTRIES TABLE
CREATE TABLE JournalEntries (
    journal_id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    title VARCHAR(255),
    content TEXT NOT NULL,
    is_private BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE
);

-- 14. GOALS TABLE
CREATE TABLE Goals (
    goal_id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    title VARCHAR(255) NOT NULL,
    description TEXT,
    target_date DATE,
    status ENUM('active', 'completed', 'abandoned') DEFAULT 'active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE
);

-- 15. EMERGENCY_CONTACTS TABLE
CREATE TABLE EmergencyContacts (
    contact_id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    name VARCHAR(200) NOT NULL,
    relationship VARCHAR(100),
    phone VARCHAR(20) NOT NULL,
    is_primary BOOLEAN DEFAULT FALSE,
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE
);

-- 16. RESOURCES TABLE
CREATE TABLE Resources (
    resource_id INT PRIMARY KEY AUTO_INCREMENT,
    title VARCHAR(255) NOT NULL,
    description TEXT,
    resource_type ENUM('article', 'video', 'audio', 'exercise', 'meditation'),
    content_url VARCHAR(500),
    category VARCHAR(100),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 17. USER_RESOURCES TABLE (Junction)
CREATE TABLE UserResources (
    user_id INT NOT NULL,
    resource_id INT NOT NULL,
    is_bookmarked BOOLEAN DEFAULT FALSE,
    completed_at TIMESTAMP,
    PRIMARY KEY (user_id, resource_id),
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (resource_id) REFERENCES Resources(resource_id) ON DELETE CASCADE
);
